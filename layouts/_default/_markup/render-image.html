{{- /* Resolve Markdown images against assets/ so they get resized to WebP like the images used in layouts. Falls back to the raw path (e.g. static/). */ -}} {{- $src := .Destination -}} {{- with resources.Get (strings.TrimPrefix "/" $src) -}} {{- $src = (.Resize "900x webp").RelPermalink -}} {{- else -}} {{- $src = $src | strings.TrimPrefix "/" | relURL -}} {{- end -}}
